- Bài 1: Tạo và thao tác với bảng
-- Tạo một bảng Employees với các cột:
-- - EmployeeID (INT, khóa chính)
-- - Name (VARCHAR(100))
-- - Age (INT)
-- - Department (VARCHAR(50))
-- - Salary (DECIMAL(10, 2))
-- Thực hiện các thao tác:
-- - Thêm 5 nhân viên với thông tin bất kỳ.
-- - Lấy tất cả thông tin của nhân viên thuộc phòng ban "IT".
-- - Cập nhật lương của nhân viên có EmployeeID = 2 thành 8,500.00.
-- - Xóa nhân viên có EmployeeID = 4.

-- 1. Tạo bảng Employees
CREATE TABLE Employees (
    EmployeeID INT PRIMARY KEY,
    Name VARCHAR(100),
    Age INT,
    Department VARCHAR(50),
    Salary DECIMAL(10, 2)
);

-- 2. Thêm 5 nhân viên vào bảng
INSERT INTO Employees (EmployeeID, Name, Age, Department, Salary) VALUES
(1, 'Nguyen Van A', 28, 'HR', 5000.00),
(2, 'Tran Thi B', 30, 'IT', 7500.00),
(3, 'Le Van C', 25, 'Marketing', 6000.00),
(4, 'Pham Thi D', 32, 'IT', 8200.00),
(5, 'Hoang Van E', 27, 'Finance', 6500.00);

-- 3. Lấy tất cả thông tin của nhân viên thuộc phòng ban "IT"
SELECT * 
FROM Employees 
WHERE Department = 'IT';

-- 4. Cập nhật lương của nhân viên có EmployeeID = 2 thành 8,500.00
UPDATE Employees 
SET Salary = 8500.00 
WHERE EmployeeID = 2;

-- 5. Xóa nhân viên có EmployeeID = 4
DELETE FROM Employees 
WHERE EmployeeID = 4;






-- Tạo bảng Sales với các cột:
-- - SaleID (INT, khóa chính)
-- - EmployeeID (INT, khóa ngoại tham chiếu đến bảng Employees.EmployeeID)
-- - SaleAmount (DECIMAL(10, 2))
-- - SaleDate (DATE)
-- Thêm dữ liệu giả lập:
-- - 5 bản ghi bán hàng với EmployeeID trùng khớp bảng Employees.
-- Thực hiện các truy vấn:
-- - Tính tổng doanh thu từ tất cả các giao dịch.
-- - Tìm doanh thu trung bình của nhân viên trong phòng ban "Sales".
-- - Liệt kê tất cả các nhân viên chưa thực hiện giao dịch bán hàng.

-- 1. Tạo bảng Sales với khóa ngoại tham chiếu đến bảng Employees
CREATE TABLE Sales (
    SaleID INT PRIMARY KEY,
    EmployeeID INT,
    SaleAmount DECIMAL(10, 2),
    SaleDate DATE,
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID)
);

-- 2. Thêm 5 bản ghi bán hàng với EmployeeID trùng khớp bảng Employees 
INSERT INTO Sales (SaleID, EmployeeID, SaleAmount, SaleDate) VALUES
(101, 1, 1500.50, '2026-06-01'),
(102, 2, 2300.00, '2026-06-02'),
(103, 1, 1200.75, '2026-06-03'),
(104, 3, 4500.00, '2026-06-04'),
(105, 5, 3100.25, '2026-06-05');

-- 3. Tính tổng doanh thu từ tất cả các giao dịch
SELECT SUM(SaleAmount) AS TotalRevenue 
FROM Sales;

-- 4. Tìm doanh thu trung bình của nhân viên trong phòng ban "Sales"
-- (Nếu trong phòng ban 'Sales' có nhân viên thực hiện giao dịch)
SELECT AVG(s.SaleAmount) AS AvgSalesDepartmentRevenue
FROM Sales s
JOIN Employees e ON s.EmployeeID = e.EmployeeID
WHERE e.Department = 'Sales';

-- 5. Liệt kê tất cả các nhân viên chưa thực hiện giao dịch bán hàng
SELECT * 
FROM Employees 
WHERE EmployeeID NOT IN (
    SELECT DISTINCT EmployeeID 
    FROM Sales
);






-- Bài 3: Thao tác với khóa ngoại và JOIN
-- Tạo bảng Projects với các cột:
-- - ProjectID (INT, khóa chính)
-- - ProjectName (VARCHAR(100))
-- - Department (VARCHAR(50))
-- Tạo bảng Assignments với các cột:
-- - AssignmentID (INT, khóa chính)
-- - EmployeeID (INT, khóa ngoại tham chiếu Employees.EmployeeID)
-- - ProjectID (INT, khóa ngoại tham chiếu Projects.ProjectID)
-- Thêm dữ liệu:
-- - 3 dự án cho bảng Projects.
-- - 5 bản ghi vào bảng Assignments.
-- Thực hiện các truy vấn:
-- - Lấy danh sách nhân viên và dự án mà họ tham gia.
-- - Liệt kê các nhân viên không tham gia dự án nào.
-- - Tìm số lượng nhân viên trong mỗi dự án.


-- 1. Tạo bảng Projects
CREATE TABLE Projects (
    ProjectID INT PRIMARY KEY,
    ProjectName VARCHAR(100),
    Department VARCHAR(50)
);

-- 2. Tạo bảng Assignments với các khóa ngoại tham chiếu
CREATE TABLE Assignments (
    AssignmentID INT PRIMARY KEY,
    EmployeeID INT,
    ProjectID INT,
    FOREIGN KEY (EmployeeID) REFERENCES Employees(EmployeeID),
    FOREIGN KEY (ProjectID) REFERENCES Projects(ProjectID)
);

-- 3. Thêm 3 dự án vào bảng Projects
INSERT INTO Projects (ProjectID, ProjectName, Department) VALUES
(1, 'Hệ thống Quản lý Nhân sự', 'HR'),
(2, 'Phát triển App Mobile', 'IT'),
(3, 'Chiến dịch Marketing Q3', 'Marketing');

-- 4. Thêm 5 bản ghi vào bảng Assignments (đảm bảo EmployeeID và ProjectID tồn tại)
INSERT INTO Assignments (AssignmentID, EmployeeID, ProjectID) VALUES
(1, 1, 1),
(2, 2, 2),
(3, 3, 3),
(4, 5, 2),
(5, 1, 3); -- Nhân viên 1 tham gia cả dự án 1 và 3

-- 5. Lấy danh sách nhân viên và dự án mà họ tham gia (dùng INNER JOIN hoặc LEFT JOIN)
SELECT e.EmployeeID, e.Name, p.ProjectID, p.ProjectName
FROM Employees e
JOIN Assignments a ON e.EmployeeID = a.EmployeeID
JOIN Projects p ON a.ProjectID = p.ProjectID;

-- 6. Liệt kê các nhân viên không tham gia dự án nào
SELECT * 
FROM Employees 
WHERE EmployeeID NOT IN (
    SELECT DISTINCT EmployeeID 
    FROM Assignments
);

-- 7. Tìm số lượng nhân viên tham gia trong mỗi dự án
SELECT p.ProjectID, p.ProjectName, COUNT(a.EmployeeID) AS NumberOfEmployees
FROM Projects p
LEFT JOIN Assignments a ON p.ProjectID = a.ProjectID
GROUP BY p.ProjectID, p.ProjectName;








-- Bài 4: Sắp xếp và lọc dữ liệu
-- Với bảng Employees, thực hiện:
-- - Lấy thông tin nhân viên có lương cao nhất.
-- - Lấy danh sách nhân viên thuộc phòng ban "HR" sắp xếp theo tuổi giảm dần.
-- - Tìm nhân viên có lương nằm trong khoảng từ 5,000.00 đến 10,000.00.
-- Với bảng Sales, thực hiện:
-- - Lấy 3 giao dịch có giá trị cao nhất.
-- - Tìm tất cả các giao dịch được thực hiện trong tháng hiện tại.



-- 1. Lấy thông tin nhân viên có lương cao nhất
SELECT * 
FROM Employees 
WHERE Salary = (SELECT MAX(Salary) FROM Employees);
-- (Hoặc có thể dùng: SELECT * FROM Employees ORDER BY Salary DESC LIMIT 1;)

-- 2. Lấy danh sách nhân viên thuộc phòng ban "HR" sắp xếp theo tuổi giảm dần
SELECT * 
FROM Employees 
WHERE Department = 'HR' 
ORDER BY Age DESC;

-- 3. Tìm nhân viên có lương nằm trong khoảng từ 5,000.00 đến 10,000.00
SELECT * 
FROM Employees 
WHERE Salary BETWEEN 5000.00 AND 10000.00;



-- 4. Lấy 3 giao dịch có giá trị cao nhất
SELECT * 
FROM Sales 
ORDER BY SaleAmount DESC 
LIMIT 3; 


-- 5. Tìm tất cả các giao dịch được thực hiện trong tháng hiện tại (Tháng 10 năm 2026)
SELECT * 
FROM Sales 
WHERE SaleDate >= '2026-10-01' AND SaleDate <= '2026-10-31';
